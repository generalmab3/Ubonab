\section{نمای کلی چارچوب}
چارچوب پیشنهادی از چهار جزء تشکیل می‌شود: داده‌های بهره‌برداری ریزشبکه، شبکه عصبی، قیود فیزیکی و فرایند بهینه‌سازی پارامترها. داده‌ها شامل متغیرهایی مانند تقاضای بار، تولید خورشیدی، قیمت برق و وضعیت شارژ هستند. شبکه عصبی بر اساس این ورودی‌ها متغیرهای تصمیم را تولید می‌کند. سپس خروجی شبکه در معادلات فیزیکی قرار می‌گیرد و باقیمانده هر معادله محاسبه می‌شود. مجموع خطای داده و خطای فیزیکی برای به‌روزرسانی پارامترهای شبکه به کار می‌رود.

\section{تعریف ورودی و خروجی مدل}
بردار ورودی نمونه در زمان $t$ را می‌توان به صورت
\begin{equation}
  \bm{x}_t=
  \left[t,\,P_{\mathrm{load}}(t),\,P_{\mathrm{PV}}(t),\,
  c_{\mathrm{buy}}(t),\,c_{\mathrm{sell}}(t),\,\mathrm{SOC}(t)\right]
  \label{eq:model-input}
\end{equation}
تعریف کرد. شبکه عصبی نگاشت $\hat{\bm{y}}_{\theta}=f_{\theta}(\bm{x})$ را یاد می‌گیرد. یک انتخاب اولیه برای خروجی عبارت است از
\begin{equation}
  \hat{\bm{y}}_t=
  \left[\hat{P}_{\mathrm{grid}}(t),\,
  \hat{P}_{\mathrm{ch}}(t),\,
  \hat{P}_{\mathrm{dis}}(t),\,
  \widehat{\mathrm{SOC}}(t+1)\right].
  \label{eq:model-output}
\end{equation}
این تعریف قطعی نیست و باید با داده موجود و هدف نهایی پروژه هماهنگ شود. برای مثال، اگر مسئله فقط پیش‌بینی وضعیت باشد، خروجی می‌تواند شامل متغیرهای حالت باشد؛ اگر هدف تولید برنامه بهره‌برداری باشد، متغیرهای تصمیم باید در خروجی قرار گیرند.

\section{باقیمانده توازن توان}
با جایگذاری خروجی شبکه در رابطه~\eqref{eq:power-balance}، باقیمانده توازن توان به صورت زیر تعریف می‌شود:
\begin{equation}
  r_{P}(t)=\hat{P}_{\mathrm{grid}}(t)+P_{\mathrm{PV}}(t)
  +\hat{P}_{\mathrm{dis}}(t)-\hat{P}_{\mathrm{ch}}(t)
  -P_{\mathrm{load}}(t).
  \label{eq:power-residual}
\end{equation}
تابع هزینه متناظر برابر است با
\begin{equation}
  \mathcal{L}_{P}=\frac{1}{N_f}\sum_{i=1}^{N_f}
  \left|r_P(t_i)\right|^2,
  \label{eq:power-loss}
\end{equation}
که $N_f$ تعداد نقاط مورد استفاده برای اعمال قانون فیزیکی است.

\section{باقیمانده وضعیت شارژ}
برای مدل گسسته، باقیمانده پویایی باتری به شکل
\begin{equation}
\begin{split}
  r_{\mathrm{SOC}}(t)=\widehat{\mathrm{SOC}}(t+1)-\mathrm{SOC}(t)
  &-\frac{\eta_{\mathrm{ch}}\hat{P}_{\mathrm{ch}}(t)\Delta t}{E_{\max}}\\
  &+\frac{\hat{P}_{\mathrm{dis}}(t)\Delta t}
  {\eta_{\mathrm{dis}}E_{\max}}
\end{split}
\label{eq:soc-residual}
\end{equation}
تعریف می‌شود. میانگین مربع این باقیمانده، مؤلفه $\mathcal{L}_{\mathrm{SOC}}$ را می‌سازد.

برای اعمال حدود متغیرها می‌توان از تابع جریمه استفاده کرد. اگر $z$ باید در بازه $[z_{\min},z_{\max}]$ باشد، جریمه زیر قابل استفاده است:
\begin{equation}
  \phi(z)=\max(0,z-z_{\max})^2+\max(0,z_{\min}-z)^2.
  \label{eq:bound-penalty}
\end{equation}
این روش یک قید نرم ایجاد می‌کند. در کاربردهایی که رعایت دقیق قید ضروری است، می‌توان تبدیل خروجی شبکه یا روش‌های بهینه‌سازی مقید را به کار گرفت.

\section{تابع هزینه نهایی}
تابع هزینه پیشنهادی از چند مؤلفه تشکیل می‌شود:
\begin{equation}
\begin{split}
  \mathcal{L}={}&
  \lambda_{d}\mathcal{L}_{\mathrm{data}}
  +\lambda_{P}\mathcal{L}_{P}
  +\lambda_{S}\mathcal{L}_{\mathrm{SOC}}\\
  &+\lambda_{c}\mathcal{L}_{\mathrm{constraint}}
  +\lambda_{e}\mathcal{L}_{\mathrm{economic}}.
\end{split}
\label{eq:total-loss}
\end{equation}
در این رابطه، $\mathcal{L}_{\mathrm{data}}$ اختلاف خروجی شبکه و داده مرجع، $\mathcal{L}_{P}$ خطای توازن توان، $\mathcal{L}_{\mathrm{SOC}}$ خطای پویایی باتری و $\mathcal{L}_{\mathrm{constraint}}$ جریمه نقض حدود است. مؤلفه اقتصادی می‌تواند هزینه خرید برق، درآمد فروش و هزینه استهلاک باتری را در بر گیرد. ضرایب $\lambda$ اهمیت نسبی هر مؤلفه را تعیین می‌کنند.

انتخاب نامناسب وزن‌ها ممکن است باعث شود یک مؤلفه بر سایر مؤلفه‌ها غالب شود. بنابراین بهتر است مقیاس متغیرها نرمال‌سازی شود و وزن‌ها با آزمایش حساسیت یا روش‌های وزن‌دهی تطبیقی تنظیم شوند. چارچوب رئیسی و همکاران نشان می‌دهد که طراحی تابع هزینه و نمونه‌برداری نقاط فیزیکی از اجزای اساسی موفقیت شبکه فیزیک‌آگاه است \pcite{raissi2019pinns}.

\section{مراحل آموزش}
فرایند آموزش را می‌توان به ترتیب زیر انجام داد:
\begin{enumerate}
  \item گردآوری یا شبیه‌سازی داده‌های بار، تولید خورشیدی، قیمت و باتری؛
  \item پاک‌سازی، تقسیم داده و نرمال‌سازی متغیرها؛
  \item مقداردهی اولیه شبکه عصبی و وزن‌های تابع هزینه؛
  \item اجرای گذر رو به جلو و تولید متغیرهای تصمیم؛
  \item محاسبه خطای داده، باقیمانده توازن، باقیمانده باتری و جریمه قیود؛
  \item انتشار معکوس و به‌روزرسانی پارامترها؛
  \item ارزیابی روی داده اعتبارسنجی و تنظیم ابرپارامترها؛
  \item گزارش نتایج روی مجموعه آزمون مستقل.
\end{enumerate}

\section{ملاحظات اعتبار علمی}
دو مقاله مبنا نقش‌های متفاوتی دارند: مقاله رئیسی و همکاران مبنای روش \lr{PINN} را فراهم می‌کند \pcite{raissi2019pinns} و مقاله اولادجو و فولی نمونه‌ای از صورت‌بندی مدیریت انرژی ریزشبکه را ارائه می‌دهد \pcite{oladejo2019microgrid}. چارچوب این فصل یک تلفیق پیشنهادی است و نباید نتایج هیچ‌یک از آن دو مقاله به‌عنوان نتیجه پروژه حاضر گزارش شود. هر جدول نتیجه، نمودار خطا یا ادعای بهبود باید مستقیماً از اجرای کد پروژه به دست آید.
