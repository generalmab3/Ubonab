\section{بردار ورودی و مرجع آموزشی}
برای ساعت $t$، بردار ورودی شامل زمان تناوبی، وضعیت روز کاری، بار، فتوولتائیک، تعرفه، وضعیت شارژ و یک پنجره کوتاه از آینده است
\begin{equation}
\begin{split}
  \bm{x}(t)=\bigl(&\sin(2\pi h/24),\,\cos(2\pi h/24),\,d,\,P_{\mathrm{L}},\,P_{\mathrm{PV}},\,c_{\mathrm{b}},\,c_{\mathrm{s}},\,s,\\
  &P_{\mathrm{L}}^{t+1},\,P_{\mathrm{PV}}^{t+1},\,P_{\mathrm{L}}^{t+3},\,P_{\mathrm{PV}}^{t+3}\bigr)
\end{split}
  \label{eq:model-input}
\end{equation}
$h$ ساعت روز و $d$ نشانگر روز کاری است. مقادیر یک و سه ساعت آینده در مدل اصلی \textbf{پیش‌بینی بی‌خطا} فرض شده‌اند؛ یعنی همان مقدار واقعی سری در اختیار مدل است. این فرض فقط برای مقایسه آفلاین با برنامه مرجع است و به معنی دسترسی بی‌خطا در میدان نیست. در فصل بعد یک آزمایش حذفی همان شبکه فیزیک‌تعبیه‌شده را بدون این چهار ورودی آموزش می‌دهد.

برنامه‌ریزی پویا با دیدن کل افق برون‌زاد، دنباله مرجع
\begin{equation}
  \bm{y}^{\star}(t)=\bigl(P_{\mathrm{G}}^{\star},\,P_{\mathrm{c}}^{\star},\,P_{\mathrm{d}}^{\star},\,s^{\star}(t+\Delta t)\bigr)
  \label{eq:reference-output}
\end{equation}
را می‌سازد. میانگین و انحراف معیار استانداردسازی فقط از مجموعه آموزش حساب می‌شوند تا آمار آزمون به آموزش نشت نکند.

\section{شبکه داده‌محور آزاد}
مدل نخست یک شبکه پیش‌خور با دو لایه پنهان ۳۲ واحدی و فعال‌ساز تانژانت هیپربولیک است. خروجی آن همان چهار مؤلفه~\eqref{eq:reference-output} را مستقیم پیش‌بینی می‌کند و تابع آموزش، میانگین مربع خطای خروجی استانداردشده است. نقش این مدل این است که نشان دهد نزدیک بودن به برچسب، به‌تنهایی به معنی قابل اجرا بودن برنامه نیست.

برای سنجش نقض‌ها، خروجی خام با \lr{SOC} پیش‌بینی‌شده خود مدل بازگشت داده می‌شود. در زنجیره \textbf{ترمیم‌شده}، که فقط برای محاسبه هزینه به کار می‌رود، \lr{SOC} هر گام از فرمان نگاشت‌یافته و دینامیک باتری دوباره حساب می‌شود نه از مؤلفه خام \lr{SOC}. فرمان خام با نگاشت ایمنی پس‌پردازشی
\begin{equation}
  u_{\mathrm{safe}}=\mathrm{clip}(\hat{P}_{\mathrm{c}}-\hat{P}_{\mathrm{d}},-P_{\max},P_{\max})
  \label{eq:usafe}
\end{equation}
به لایه~\eqref{eq:hard-soc} داده می‌شود. عملگر گیره $\mathrm{clip}(x,a,b)=\min(\max(x,a),b)$ مقدار را در بازه $[a,b]$ نگه می‌دارد. هزینه حاصل با برچسب «داده‌محور با نگاشت ایمنی» گزارش می‌شود و نباید با هزینه خروجی خام احتمالاً ناممکن یکی گرفته شود.

\section{شبکه فیزیک‌تعبیه‌شده}
شبکه دوم همان عرض و عمق لایه‌های پنهان را دارد، اما به‌جای چهار خروجی آزاد فقط یک عدد $z(t)$ می‌سازد. فرمان علامت‌دار باتری از رابطه
\begin{equation}
  u(t)=P_{\max}\tanh\bigl(z(t)\bigr)
  \label{eq:signed-command}
\end{equation}
به‌دست می‌آید. علامت مثبت شارژ و علامت منفی دشارژ است. چون بازه تانژانت هیپربولیک در $\pm 1$ باز است، فرمان دقیقاً $\pm P_{\max}$ نمی‌شود و فقط به آن نزدیک می‌گردد. سقف وابسته به \lr{SOC} چنین است
\begin{equation}
\begin{aligned}
  \bar{P}_{\mathrm{c}}(s)&=\max\left(0,\min\left(P_{\max},\frac{(s_{\max}-s)E_{\max}}{\eta_{\mathrm{c}}\Delta t}\right)\right),\\
  \bar{P}_{\mathrm{d}}(s)&=\max\left(0,\min\left(P_{\max},\frac{(s-s_{\min})E_{\max}\eta_{\mathrm{d}}}{\Delta t}\right)\right)
\end{aligned}
  \label{eq:soc-caps}
\end{equation}
سپس توان شارژ و دشارژ با بریدن فرمان روی این سقف‌ها به‌دست می‌آید
\begin{equation}
  P_{\mathrm{c}}=\min\bigl([u]_{+},\bar{P}_{\mathrm{c}}(s)\bigr),\qquad
  P_{\mathrm{d}}=\min\bigl([-u]_{+},\bar{P}_{\mathrm{d}}(s)\bigr)
  \label{eq:hard-powers}
\end{equation}
و
\begin{equation}
\begin{aligned}
  s(t+\Delta t)&=s(t)+\frac{\eta_{\mathrm{c}}P_{\mathrm{c}}\Delta t}{E_{\max}}-\frac{P_{\mathrm{d}}\Delta t}{\eta_{\mathrm{d}}E_{\max}},\\
  P_{\mathrm{G}}&=P_{\mathrm{L}}+P_{\mathrm{c}}-P_{\mathrm{PV}}-P_{\mathrm{d}}
\end{aligned}
  \label{eq:hard-soc}
\end{equation}
توان شبکه از توازن~\eqref{eq:power-balance} پس از تعیین فرمان باتری بسته می‌شود. در این مطالعه سقف واردات یا صادرات برای $P_{\mathrm{G}}$ گذاشته نشده است؛ بنابراین توازن همواره با تبادل شبکه برقرار می‌ماند.

از~\eqref{eq:hard-powers} در هر گام حداکثر یکی از $P_{\mathrm{c}}$ و $P_{\mathrm{d}}$ مثبت است. سقف‌های~\eqref{eq:soc-caps} با $\max(0,\cdot)$ نامنفی می‌مانند حتی اگر $s$ به‌خاطر گرد کردن اندکی از بازه خارج شود. حدود \lr{SOC} و توازن توان نیز از~\eqref{eq:soc-caps} و~\eqref{eq:hard-soc} برقرار می‌شوند. پس باقیمانده~\eqref{eq:power-residual} و دینامیک \lr{SOC}، جز خطای گرد کردن، باید صفر باشند؛ به شرط آن‌که وضعیت شارژ آغازین در $[s_{\min},s_{\max}]$ باشد و در آزمون بازگشتی نیز $s$ از همین نگاشت به‌روز شود. تابع آموزش، خطای فرمان مرجع است
\begin{equation}
  \mathcal{L}_{u}=\frac{1}{N}\sum_{i=1}^{N}\left(\tanh(z_{i})-\frac{u_{i}^{\star}}{P_{\max}}\right)^{2}
  \label{eq:command-loss}
\end{equation}
از نظر مفهومی این ساختار با یادگیری فیزیک‌آگاه هم‌خوان است، اما نام دقیق آن در این گزارش فیزیک‌تعبیه‌شده با قید سخت است. برتری آن نسبت به جریمه نرم این است که برای قیود اساسی به وزن جریمه نیاز نیست. ترتیب محاسبات در پیوست~\ref{app:hard-layer} آمده است.

\section{تنظیم آموزش و افراز داده}
نمونه‌ها به ترتیب زمان می‌مانند: ۶۰ درصد اول آموزش، ۲۰ درصد بعدی اعتبارسنجی و ۲۰ درصد پایانی آزمون است. شبکه‌ها با بهینه‌ساز آدام \pcite{kingma2015adam} آموزش می‌بینند. هر \textbf{دسته} ۶۴ نمونه است و حداکثر ۴۵۰ \textbf{دوره}، یعنی ۴۵۰ عبور کامل از مجموعه آموزش، اجرا می‌شود. \textbf{توقف زودهنگام} یعنی اگر خطای اعتبارسنجی برای ۵۵ دوره بهتر نشود، آموزش قطع شود؛ داده آزمون در این انتخاب دخالت ندارد. پنج بذر ۷، ۱۱، ۱۹، ۲۳ و ۲۹ فقط مقداردهی اولیه و ترتیب دسته‌ها را عوض می‌کنند. بنابراین انحراف معیار جدول، پراکندگی آموزش است نه تغییر داده یا تغییر تقسیم. در آموزش، $s$ از مسیر مرجع می‌آید و در آزمون بازگشتی از مسیر خود مدل؛ بخشی از شکاف هزینه می‌تواند از همین ناهماهنگی توزیع باشد، نه فقط از خطای یک‌گامی.

\section{شاخص‌های ارزیابی}
میانگین مربع خطا (\lr{MSE}) تجمیعی میانگین روی چهار مؤلفه~\eqref{eq:reference-output} است
\begin{equation}
  \mathrm{MSE}_{\mathrm{comp}}=\frac{1}{4N}\sum_{i=1}^{N}\sum_{j=1}^{4}\bigl(\hat{y}_{ij}-y_{ij}^{\star}\bigr)^{2}
  \label{eq:mse-comp}
\end{equation}
توان و \lr{SOC} هم‌واحد نیستند؛ بنابراین این عدد معیار فیزیکی اصلی نیست و در جدول دقت فصل بعد نمی‌آید. مقدار آن در \lr{metrics.json} و پیوست~\ref{app:reproducibility} بایگانی شده است. مقایسه دقت با میانگین قدر مطلق خطا برای توان شبکه، فرمان باتری و \lr{SOC} انجام می‌شود. در مدل فیزیک‌تعبیه‌شده این چهار مؤلفه پس از لایه سخت ساخته می‌شوند و تابع آموزش همان~\eqref{eq:command-loss} است، نه این میانگین چهارتایی.

\textbf{امکان‌پذیری} با پنج شاخص سنجیده می‌شود: میانگین $|r_{\mathrm{G}}|$، نرخ نقض \lr{SOC}، نرخ توان شارژ منفی، نرخ توان دشارژ منفی و نرخ شارژ و دشارژ هم‌زمان. برای هر برنامه قابل اجرا، هزینه کل آزمون از جمع~\eqref{eq:step-cost} به‌دست می‌آید. مرجع برنامه‌ریزی پویا یک کران پایین تفسیری برای هزینه است. حالت بدون باتری و سیاست حریصانه دو کران بالای ساده‌اند: اولی باتری را به‌کار نمی‌برد و دومی فقط هزینه همان ساعت را روی فرمان مجاز کمینه می‌کند.
