\setlength{\parskip}{0pt}
\setlength{\emergencystretch}{2em}

% شماره صفحه در وسط پایین صفحه؛ سبک plain استاندارد دقیقاً همین چینش را دارد.
% استفاده نکردن از fancyhdr، سازگاری بهتری با bidi و xepersian ایجاد می‌کند.
\pagestyle{plain}

% قالب عنوان فصل و بخش‌ها
\titleformat{\chapter}[display]
  {\centering\TitleFont\bfseries\fontsize{16pt}{24pt}\selectfont}
  {فصل \thechapter}
  {0.5em}
  {}
\titleformat{name=\chapter,numberless}[display]
  {\centering\TitleFont\bfseries\fontsize{16pt}{24pt}\selectfont}
  {}
  {0pt}
  {}
\titlespacing*{\chapter}{0pt}{-10pt}{28pt}

\titleformat{\section}
  {\HeadingFont\bfseries\itshape\fontsize{14pt}{21pt}\selectfont}
  {\thesection}
  {0.75em}
  {}
\titlespacing*{\section}{0pt}{21pt}{7pt}

\titleformat{\subsection}
  {\HeadingFont\bfseries\itshape\fontsize{14pt}{21pt}\selectfont}
  {\thesubsection}
  {0.75em}
  {}
\titlespacing*{\subsection}{0pt}{14pt}{5pt}

\titleformat{\subsubsection}
  {\HeadingFont\bfseries\fontsize{13pt}{19pt}\selectfont}
  {\thesubsubsection}
  {0.75em}
  {}

% جداکننده شماره فصل، بخش، شکل، جدول و رابطه مطابق دستورالعمل دانشگاه.
\SepMark{-}

% bidi ترتیب اجزای شماره را در متن راست‌به‌چپ معکوس می‌کند. برای آنکه خروجی
% چاپی به شکل «شماره فصل-شماره بخش» دیده شود، ترتیب داخلی شمارنده‌ها برعکس
% تعریف می‌شود. این فرمان باید پس از \begin{document} اجرا شود.
\makeatletter
\newcommand{\ApplyBonabNumbering}{%
  \renewcommand{\thesection}{\arabic{section}\@SepMark\thechapter}%
  \renewcommand{\thesubsection}{\arabic{subsection}\@SepMark\arabic{section}\@SepMark\thechapter}%
  \renewcommand{\thesubsubsection}{\arabic{subsubsection}\@SepMark\arabic{subsection}\@SepMark\arabic{section}\@SepMark\thechapter}%
  \renewcommand{\thefigure}{\arabic{figure}\@SepMark\thechapter}%
  \renewcommand{\thetable}{\arabic{table}\@SepMark\thechapter}%
  \renewcommand{\theequation}{\arabic{equation}\@SepMark\thechapter}%
  \renewcommand{\thedefinition}{\arabic{definition}\@SepMark\thechapter}%
}
\makeatother

% فرمول‌ها از حاشیه چپ آغاز می‌شوند و شماره رابطه در سمت راست قرار می‌گیرد.
\setlength{\mathindent}{0pt}

% عنوان شکل‌ها و جدول‌ها با اندازه ۱۰ پوینت مطابق دستورالعمل دانشگاه
\DeclareCaptionFont{bonabcaption}{\fontsize{10pt}{15pt}\selectfont}
\captionsetup{
  font={bonabcaption,stretch=1.15},
  labelfont=bf,
  justification=centering,
  singlelinecheck=false,
  labelsep=period
}
\captionsetup[table]{position=top,skip=7pt}
\captionsetup[figure]{position=bottom,skip=7pt}

% فهرست‌ها
% از گلوله ریاضی استفاده می‌شود تا نبود نویسه bullet در B Mitra مربع نسازد.
\setlist[itemize]{label={\ensuremath{\bullet}},topsep=4pt,itemsep=2pt,parsep=0pt}
\setlist[enumerate]{topsep=4pt,itemsep=2pt,parsep=0pt}
